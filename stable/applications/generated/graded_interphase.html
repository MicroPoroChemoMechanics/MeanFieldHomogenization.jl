Layered inclusions, for the layered sphere and graded_layers, and Layered spheres for the recurrence behind it.

Around a particle the matrix is often altered over a zone where its moduli vary continuously: a diffusion zone around a fiber, the transition zone around an aggregate in concrete. This page enters such a zone as thin uniform layers, measures the error of the layers against the exact solution of Lutz and Zimmerman for the bulk modulus and for the conductivity, and adds the shear modulus, which the layers give as well.

The layers ​

The profile reaches the matrix at infinity. It is cut at the radius  , where it differs from the matrix by , about . graded_layers then cuts the zone at given radii and gives each layer the moduli of the profile at its mid-radius. The radii are taken here in equal steps of , which makes the layers thin near the particle, where the profile is steep.

julia
const r_out = 5a
edges(n) = a ./ range(1, a / r_out; length = n + 1)
function particle(D, n)
    radii, moduli = graded_layers(zone(D), edges(n))
    return LayeredSphere((a, radii...), (C_i, moduli...))
end
layer_count(particle(0.5, 60))                 ## the particle and 60 layers

function frobenius(K_i, λ_if, μ_if, λ_m, μ_m, b; terms = 400)
    M_m, M_if = λ_m + 2μ_m, λ_if + 2μ_if
    ΔM, Δλ = M_if - M_m, λ_if - λ_m
    ratio(n) = -(ΔM * (n^2 + (b - 3) * n - b) - 2b * Δλ) / ((n + b) * (n + b - 3) * M_m)
    S₁ = S₁′ = S₂ = S₂′ = 0.0
    c₁ = c₂ = 1.0
    for j in 0:terms
        S₁ += c₁; S₁′ += j * b * c₁; S₂ += c₂; S₂′ += (j * b + 3) * c₂
        c₁ *= ratio(j * b); c₂ *= ratio(j * b + 3)
    end
    K_if = λ_if + 2μ_if / 3
    return (3 * (K_i - K_if) * S₁ + M_if * S₁′) / (3 * (K_i - K_if) * S₂ + M_if * S₂′)
end
f_exact(D) = frobenius(K_i, (1 - D) * (K_m - 2μ_m / 3), (1 - D) * μ_m, K_m - 2μ_m / 3, μ_m, b)

julia
const k_m, k_i = 1.0, 5.0
function f_conduction(D, n)
    radii, moduli = graded_layers(r -> TensISO{3}(k_m * (1 - D * (a / r)^b)), edges(n))
    sphere = LayeredSphere((a, radii...), (TensISO{3}(k_i), moduli...))
    n_k = canonical_components(property_contribution(sphere, TensISO{3}(k_m), TensISO{3}(k_m)))[1, 1]
    return n_k / (3k_m) * (r_out / a)^3
end
f_conduction_exact(D) = frobenius(k_i / 3, 0.0, (1 - D) * k_m / 2, 0.0, k_m / 2, b)
